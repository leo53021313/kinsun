# 網頁版全功能前端 P4（通知連動與收尾）實作計畫

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 兩欄活起來——家屬在右邊設一個提醒，長輩左邊的手機外框頂端滑進一張模擬的系統通知；長輩對麥克風講出危急語句，家屬右邊滑進危急警報。加上端到端測試與跨瀏覽器驗收，整個網頁版到此完成。

**Architecture:** 通知的資料來源是**既有的**兩支 App 內通知端點（後端一行都不用改——危急事件本來就會寫進 `app_notifications`）。輪詢負責「後端主動產生的東西」（危急警報、排程提醒），跨欄 event bus 負責「這個頁面自己剛做的事」（家屬新增排程後長輩端立刻重載），兩者互補。

**Tech Stack:** React 19 · Playwright（E2E）

**前置條件：** P1、P2、P3 全部完成並驗收過。
**設計依據：** `docs/superpowers/specs/2026-07-30-web-demo-design.md` §9（兩端連動與模擬通知）、§11（測試）、§12（已知風險）。

```tsx
import { render, screen } from "@testing-library/react";
import userEvent from "@testing-library/user-event";
import { describe, expect, it, vi } from "vitest";

import { detectOs } from "./osStyle";

const ITEM = { id: "n1", title: "金孫", content: "提醒您：降血壓藥", at: 1754000000 };

describe("NotificationBanner", () => {
  it("沒有東西時不畫任何東西", () => {
    const { container } = render(
      <NotificationBanner item={null} os="ios" onDismiss={vi.fn()} />,
    );
    expect(container).toBeEmptyDOMElement();
  });

  it("顯示標題與內容", () => {
    render(<NotificationBanner item={ITEM} os="ios" onDismiss={vi.fn()} />);
    expect(screen.getByText("金孫")).toBeInTheDocument();
    expect(screen.getByText("提醒您：降血壓藥")).toBeInTheDocument();

    const onDismiss = vi.fn();
    render(<NotificationBanner item={ITEM} os="ios" onDismiss={onDismiss} />);
    await userEvent.click(screen.getByRole("button", { name: "關閉" }));

    const { rerender } = render(
      <NotificationBanner item={ITEM} os="ios" onDismiss={vi.fn()} />,

    rerender(<NotificationBanner item={ITEM} os="android" onDismiss={vi.fn()} />);
```

```tsx
/**
 * 模擬的系統通知橫幅（W-13）。
 *
 * ⚠️ 這是**畫面上的模擬**，不是瀏覽器的 Notification API：後端只支援 Expo 推播
 * （`push_tokens.py` 的平台白名單只有 android／ios），網頁沒有那條路，而做 Web
 * Push 是一整包新的後端工作（VAPID＋Service Worker＋金鑰管理）。展示要的正是
 * 「手機上跳出通知」的那個畫面感，模擬完全足夠。
 *
 * ⚠️ 用 `role="status"`：通知的重點就是「打斷你、讓你知道」。只有視覺效果的話，
 * 對用讀螢幕的人等於這則通知不存在。
 */

import { strings } from "@/strings";

import type { PhoneOs } from "./osStyle";

export type BannerItem = {
  id: string;
  title: string;
  content: string;
  at: number;
};

const STYLE: Record<PhoneOs, string> = {
  // iOS：毛玻璃、大圓角、置中偏上。
  ios: "rounded-3xl bg-white/80 backdrop-blur-md shadow-lg ring-1 ring-black/5",
  // Android：Material 卡片、較小圓角、實色。
  android: "rounded-2xl bg-surface shadow-xl border border-line",
};

export function NotificationBanner(props: {
  item: BannerItem | null;
  os: PhoneOs;
  onDismiss: () => void;
}) {
  const { item, os, onDismiss } = props;
  if (item === null) {
    return null;
  }
  return (
    <div
      // ⚠️ key 用 item.id：沒有它的話 React 會沿用同一個 DOM 節點，第二則會靜靜地
      // 換掉字、完全沒有「又來一則」的感覺——而那正是展示要的效果。
      key={item.id}
      role="status"
      className={`pointer-events-auto flex animate-[slideDown_240ms_ease-out] items-start gap-2 p-3 ${STYLE[os]}`}
    >
      <div className="min-w-0 flex-1">
        <p className="text-xs font-bold text-ink">{item.title}</p>
        <p className="mt-0.5 break-words text-sm leading-5 text-ink">{item.content}</p>
      </div>
      <button
        type="button"
        aria-label={strings.common.close}
        onClick={onDismiss}
        className="shrink-0 rounded-full px-2 text-lg leading-none text-ink-soft"
      >
        ×
      </button>
    </div>
  );
}
```

```ts
function item(at: number, content = "x"): AppNotification {
  return { content, created_at: at };

describe("pickNewItems", () => {

    expect(pickNewItems([item(300), item(200), item(100)], 150)).toEqual([item(300), item(200)]);
```

```ts
/**
 * 通知輪詢：把後端主動產生的東西（危急警報、排程提醒）變成畫面上的橫幅。
 *
 * ⚠️ 為什麼是輪詢而不是 WebSocket：長輩端的 WS 是對講機專用的，家屬端根本沒有
 * 連線。替家屬端另開一條下行通道是後端的新工作，而這個規模的併發（展示現場
 * 十幾個人）用兩秒輪詢完全吃得住。
 *
 * ⚠️ **第一次載入不補播歷史**：那是展示現場最尷尬的失敗——一進站就滑進十幾張
 * 橫幅、蓋滿整個手機。第一次的用途是「記下目前的水位」。
 */

import type { AppNotification } from "kinsun-shared/types";
import { useCallback, useEffect, useRef, useState } from "react";

import { listElderNotifications } from "@/elder/api";
import { listNotifications } from "@/guardian/api";

import type { BannerItem } from "./NotificationBanner";
import { loadSeenAt, saveSeenAt, type Audience } from "./seen";

const DEFAULT_INTERVAL_MS = 2000;

/** 純函式：挑出比水位新的那些。水位為 0（從未看過）時一律不算新的。 */
export function pickNewItems(items: AppNotification[], sinceAt: number): AppNotification[] {
  if (sinceAt === 0) {
    return [];
  }
  return items
    .filter((item) => item.created_at > sinceAt)
    .sort((a, b) => b.created_at - a.created_at);
}

const TITLE: Record<Audience, string> = {
  guardian: "金孫",
  elder: "金孫",
};

export function useNotificationFeed(options: {
  audience: Audience;
  token: string;
  intervalMs?: number;
  /** 值一變就立刻重拉一次（跨欄連動用，見 notify/bus）。 */
  reloadSignal?: number;
}) {
  const { audience, token, intervalMs = DEFAULT_INTERVAL_MS, reloadSignal = 0 } = options;
  const [banner, setBanner] = useState<BannerItem | null>(null);
  const [unread, setUnread] = useState(0);
  // 佇列而非直接覆寫：一次輪詢可能拿到兩則新的（排程提醒剛好與危急警報同時），
  // 直接覆寫會讓第一則在畫面上一閃而過。
  const queue = useRef<BannerItem[]>([]);
  const seenAt = useRef(0);

  const shift = useCallback(() => {
    setBanner(queue.current.shift() ?? null);
  }, []);

  const poll = useCallback(async () => {
    if (!token) return;
    const items =
      audience === "guardian"
        ? await listNotifications(token)
        : await listElderNotifications(token);
    const fresh = pickNewItems(items, seenAt.current);
    if (items.length > 0) {
      const newest = Math.max(...items.map((item) => item.created_at));
      seenAt.current = Math.max(seenAt.current, newest);
      saveSeenAt(seenAt.current, audience);
    }
    setUnread(items.filter((item) => item.created_at > loadSeenAt(audience)).length);
    if (fresh.length > 0) {
      // 最舊的先播——事情發生的順序才是使用者理解得了的順序。
      queue.current.push(
        ...fresh
          .slice()
          .reverse()
          .map((item) => ({
            id: `${item.created_at}-${item.content}`,
            title: TITLE[audience],
            content: item.content,
            at: item.created_at,
          })),
      );
      setBanner((current) => current ?? queue.current.shift() ?? null);
    }
  }, [audience, token]);

  useEffect(() => {
    seenAt.current = loadSeenAt(audience);
  }, [audience]);

  useEffect(() => {
    if (!token) return;
    let alive = true;
    const run = () => {
      // 輪詢失敗完全靜默：通知是加分項，網路抖一下不該在畫面上留紅字。
      void poll().catch(() => undefined);
    };
    run();
    const timer = setInterval(() => alive && run(), intervalMs);
    return () => {
      alive = false;
      clearInterval(timer);
    };
  }, [poll, token, intervalMs, reloadSignal]);

  // 橫幅自動消失。3.5 秒足夠看完一句話，又不會擋住底下的操作太久。
  useEffect(() => {
    if (banner === null) return;
    const timer = setTimeout(shift, 3500);
    return () => clearTimeout(timer);
  }, [banner, shift]);

  return { banner, unread, dismiss: shift, reload: () => void poll().catch(() => undefined) };
}
```

```ts
/**
 * 跨欄事件匯流排。
 *
 * ⚠️ 為什麼需要它：輪詢負責「後端主動產生的東西」（危急警報、排程提醒），但家屬
 * 剛按下的那個新增排程，長輩端要等最多兩秒才看得到——而展示時那兩秒剛好落在
 * 「你看，左邊出現了」這句話的中間。兩欄在同一個 JS 環境裡，直接通知一聲就好。
 *
 * 刻意做得極小：它不搬運資料，只說「有事發生了，你自己去拉」。搬資料的話兩欄
 * 就得共用型別與快取，而那正是把兩個獨立的畫面黏成一團的第一步。
 */

import { useEffect, useState } from "react";

export type StageEvent = "guardian-wrote" | "elder-talked";

const listeners = new Map<StageEvent, Set<() => void>>();

export function emitStageEvent(event: StageEvent): void {
  listeners.get(event)?.forEach((listener) => listener());
}

/** 回一個每次事件都會變的計數；直接餵給 `useNotificationFeed` 的 `reloadSignal`。 */
export function useStageEvent(event: StageEvent): number {
  const [tick, setTick] = useState(0);
  useEffect(() => {
    const listener = () => setTick((value) => value + 1);
    const set = listeners.get(event) ?? new Set();
    set.add(listener);
    listeners.set(event, set);
    return () => {
      set.delete(listener);
    };
  }, [event]);
  return tick;
}
```

```tsx
/**
 * 雙欄舞台（spec §5.3、§9）。
 *
 * ⚠️ 分成 StagePage（掛 Provider）與 StageBody（用 Provider）兩層：session 的讀取
 * 必須發生在 Provider 內部。合成一層的話 useSession 會擲出「必須在 Provider 之內
 * 使用」。
 *
 * ⚠️ 兩個 Provider 都掛在最外層而不是各掛在自己那一欄裡面：跨欄連動需要兩邊在
 * 同一棵樹底下。
 */

import { useState } from "react";

import { NotificationBanner } from "@/notify/NotificationBanner";
import { useStageEvent } from "@/notify/bus";
import { detectOs, type PhoneOs } from "@/notify/osStyle";
import { useNotificationFeed } from "@/notify/useNotificationFeed";
import { ElderSession, GuardianSession } from "@/session/contexts";
import { strings } from "@/strings";

import { ElderApp } from "@/elder/ElderApp";
import { GuardianApp } from "@/guardian/GuardianApp";

import { PhoneFrame } from "./PhoneFrame";

type Pane = "elder" | "guardian";

export function StagePage() {
  return (
    <ElderSession.Provider>
      <GuardianSession.Provider>
        <StageBody />
      </GuardianSession.Provider>
    </ElderSession.Provider>
  );
}

function StageBody() {
  const [pane, setPane] = useState<Pane>("elder");
  const [os, setOs] = useState<PhoneOs>(() => detectOs(navigator.userAgent));

  const elder = ElderSession.useSession();
  const guardian = GuardianSession.useSession();

  // 家屬寫入之後立刻叫長輩端重拉，不等下一次輪詢（見 notify/bus 的說明）。
  const guardianWrote = useStageEvent("guardian-wrote");

  const elderFeed = useNotificationFeed({
    audience: "elder",
    token: elder.session?.token ?? "",
    reloadSignal: guardianWrote,
  });
  const guardianFeed = useNotificationFeed({
    audience: "guardian",
    token: guardian.session?.token ?? "",
  });

  return (
    <main className="min-h-dvh bg-background p-4 lg:p-8">
      <div className="mx-auto mb-4 flex w-full max-w-5xl items-center justify-between gap-2">
        {/* 窄螢幕的切換頁籤。寬螢幕兩欄都看得到，不需要它。 */}
        <div role="tablist" className="flex flex-1 gap-2 lg:hidden">
          {(
            [
              ["elder", strings.stage.elderTab],
              ["guardian", strings.stage.guardianTab],
            ] as const
          ).map(([value, label]) => (
            <button
              key={value}
              role="tab"
              aria-selected={pane === value}
              onClick={() => setPane(value)}
              className={`min-h-12 flex-1 rounded-2xl text-base font-bold transition-colors ${
                pane === value
                  ? "bg-primary text-white"
                  : "border border-line bg-surface text-ink-soft"
              }`}
            >
              {label}
            </button>
          ))}
        </div>
        {/* UA 只能給一個合理的預設，而展示時觀眾會想看兩種、投影用的筆電也只有
            一種 UA。 */}
        <button
          type="button"
          aria-label={strings.stage.notificationStyle(os)}
          onClick={() => setOs((current) => (current === "ios" ? "android" : "ios"))}
          className="min-h-12 shrink-0 rounded-2xl border border-line bg-surface px-4 text-sm font-semibold text-ink-soft"
        >
          {strings.stage.notificationStyle(os)}
        </button>
      </div>

      <div className="mx-auto grid w-full max-w-5xl gap-8 lg:grid-cols-2">
        <div className={pane === "elder" ? "" : "hidden lg:block"}>
          <PhoneFrame
            title={strings.stage.elderTitle}
            os={os}
            notificationSlot={
              <NotificationBanner item={elderFeed.banner} os={os} onDismiss={elderFeed.dismiss} />
            }
          >
            <ElderApp />
          </PhoneFrame>
        </div>
        <div className={pane === "guardian" ? "" : "hidden lg:block"}>
          <PhoneFrame
            title={strings.stage.guardianTitle}
            os={os}
            notificationSlot={
              <NotificationBanner
                item={guardianFeed.banner}
                os={os}
                onDismiss={guardianFeed.dismiss}
              />
            }
          >
            <GuardianApp />
          </PhoneFrame>
        </div>
      </div>
    </main>
  );
}
```
